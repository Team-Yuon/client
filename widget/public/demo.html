⚙️ 위젯 설정

📝 사용 방법

<!-- 1. 스크립트 로드 -->
<script src="https://your-cdn.com/yuon-widget.js"></script>

<!-- 2. 초기화 -->
<script>
  YuonChat.init({
    apiUrl: 'https://api.yourdomain.com/api/v1',
    wsUrl: 'wss://api.yourdomain.com/api/v1/ws',
    theme: 'dark',
    position: 'right',
    primaryColor: '#3b82f6',
    welcomeMessage: '안녕하세요! 무엇을 도와드릴까요?'
  })
</script>
